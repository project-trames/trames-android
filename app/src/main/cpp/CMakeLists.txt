cmake_minimum_required(VERSION 3.22.1)
project(trames_android LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
if(ANDROID_ABI STREQUAL "arm64-v8a")
    set(RUST_TARGET aarch64-linux-android)
elseif(ANDROID_ABI STREQUAL "x86_64")
    set(RUST_TARGET x86_64-linux-android)
else()
    message(FATAL_ERROR "Unsupported packaging ABI: ${ANDROID_ABI}")
endif()
get_filename_component(PROJECT_ROOT "${CMAKE_CURRENT_LIST_DIR}/../../../.." ABSOLUTE)
set(RUST_ARCHIVE "${PROJECT_ROOT}/target/${RUST_TARGET}/release/libtrames_android_core.a")
if(NOT EXISTS "${RUST_ARCHIVE}")
    message(FATAL_ERROR "Build the Rust core first: scripts/build.ps1")
endif()
add_library(trames_core STATIC IMPORTED)
set_target_properties(trames_core PROPERTIES IMPORTED_LOCATION "${RUST_ARCHIVE}")
add_library(trames_android SHARED bridge.cpp)
target_link_libraries(trames_android PRIVATE trames_core android log dl m)
target_link_options(trames_android PRIVATE "-Wl,-z,max-page-size=16384" "-Wl,-z,common-page-size=16384" "-Wl,--gc-sections")
